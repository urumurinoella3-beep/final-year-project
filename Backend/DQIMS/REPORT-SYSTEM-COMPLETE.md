# Report System - Complete with Excel & Word + Activity Logs ✅

## Overview

The DQIMS report system now generates professional reports in **both Excel and Word formats**, each with comprehensive **activity logs** for audit purposes. CSV has been removed as requested.

## What's Available

### ✅ Excel Reports (with Activity Logs)
- **4 Professional Sheets:**
  1. Summary - Metadata, filters, statistics
  2. Issues - Complete data listing
  3. Statistics - Breakdowns by status, priority, severity, department
  4. **Activity Log** - Complete audit trail with urgent issue highlighting

### ✅ Word Reports (with Activity Logs)  
- **6 Professional Sections:**
  1. Title Page - RRA branding, metadata, filters
  2. Executive Summary - Key statistics
  3. Detailed Statistics - Multiple breakdowns
  4. Issues Table - Complete listing
  5. **Activity Log** - Complete audit trail with urgent issues marked
  6. Footer - Confidentiality notice

### ❌ CSV Removed
- Deleted as requested
- Excel and Word serve all reporting needs

## API Endpoints

### 1. Dashboard Statistics
```
GET /api/v1/reports/dashboard-stats
Authorization: Bearer {token}
```

**Response:**
```json
{
  "totalIssues": 150,
  "openIssues": 45,

  "resolvedIssues": 50,
  "closedIssues": 25,
  "totalUsers": 42
}
```

```
GET /api/v1/reports/generate-excel
GET /api/v1/reports/generate-excel?department=Finance

Authorization: Bearer {token}
```

**Response:** Excel file (.xlsx) with 4 sheets including Activity Log

### 3. Generate Word Report (with Activity Log)
```
GET /api/v1/reports/generate-word
GET /api/v1/reports/generate-word?department=Finance

GET /api/v1/reports/generate-word?startDate=2024-01-01T00:00:00&endDate=2024-12-31T23:59:59
Authorization: Bearer {token}
```

**Response:** Word file (.docx) with 6 sections including Activity Log

## Frontend Integration

### Download Excel Report
```javascript
const downloadExcelReport = async (filters = {}) => {
  const params = new URLSearchParams();
  if (filters.department) params.append('department', filters.department);
  if (filters.status) params.append('status', filters.status);
  if (filters.startDate) params.append('startDate', filters.startDate);
  if (filters.endDate) params.append('endDate', filters.endDate);
  
  const response = await fetch(
    `http://localhost:8080/api/v1/reports/generate-excel?${params.toString()}`,
    {
      headers: {
        'Authorization': `Bearer ${token}`
      }
    }
  );
  
  const blob = await response.blob();
  const url = window.URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `DQIMS-Report-${new Date().toISOString().split('T')[0]}.xlsx`;
  document.body.appendChild(a);
  a.click();
  window.URL.revokeObjectURL(url);
  document.body.removeChild(a);
};
```

### React Component Example
```typescript
import React, { useState } from 'react';

const ReportDownloader: React.FC = () => {
  const [loading, setLoading] = useState(false);
  const [format, setFormat] = useState<'excel' | 'word'>('excel');
  const [filters, setFilters] = useState({
    department: '',
    status: '',
    startDate: '',
    endDate: ''
  });

  const handleDownload = async () => {
    setLoading(true);
    
    const params = new URLSearchParams();
    if (filters.department) params.append('department', filters.department);
    if (filters.status) params.append('status', filters.status);
    if (filters.startDate) params.append('startDate', filters.startDate + ':00');
    if (filters.endDate) params.append('endDate', filters.endDate + ':00');
    
    const endpoint = format === 'excel' ? 'generate-excel' : 'generate-word';
    const extension = format === 'excel' ? 'xlsx' : 'docx';
    
    try {
      const response = await fetch(
        `http://localhost:8080/api/v1/reports/${endpoint}?${params.toString()}`,
        {
          headers: {
            'Authorization': `Bearer ${localStorage.getItem('token')}`
          }
        }
      );
      
      if (!response.ok) throw new Error('Report generation failed');
      
      const blob = await response.blob();
      const url = window.URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `DQIMS-Report-${new Date().toISOString().split('T')[0]}.${extension}`;
      document.body.appendChild(a);
      a.click();
      window.URL.revokeObjectURL(url);
      document.body.removeChild(a);
      
      alert('Report downloaded successfully!');
    } catch (error) {
      console.error('Download failed:', error);
      alert('Failed to download report. Please try again.');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="report-downloader">
      <h2>Generate Report with Activity Log</h2>
      
      <div className="format-selector">
        <label>
          <input 
            type="radio" 
            value="excel" 
            checked={format === 'excel'} 
            onChange={(e) => setFormat('excel')}
          />
          Excel Report
        </label>
        <label>
          <input 
            type="radio" 
            value="word" 
            checked={format === 'word'} 
            onChange={(e) => setFormat('word')}
          />
          Word Report
        </label>
      </div>
      
      <div className="filters">
        <select 
          value={filters.department} 
          onChange={(e) => setFilters({...filters, department: e.target.value})}
        >
          <option value="">All Departments</option>
          <option value="Finance">Finance</option>
          <option value="IT">IT</option>
          <option value="HR">HR</option>
          <option value="Operations">Operations</option>
        </select>
        
        <select 
          value={filters.status} 
          onChange={(e) => setFilters({...filters, status: e.target.value})}
        >
          <option value="">All Statuses</option>
          <option value="OPEN">Open</option>
          <option value="IN_PROGRESS">In Progress</option>
          <option value="RESOLVED">Resolved</option>
          <option value="CLOSED">Closed</option>
        </select>
        
        <input 
          type="datetime-local" 
          value={filters.startDate} 
          onChange={(e) => setFilters({...filters, startDate: e.target.value})}
          placeholder="Start Date"
        />
        
        <input 
          type="datetime-local" 
          value={filters.endDate} 
          onChange={(e) => setFilters({...filters, endDate: e.target.value})}
          placeholder="End Date"
        />
      </div>
      
      <button 
        onClick={handleDownload} 
        disabled={loading}
        className="download-btn"
      >
        {loading ? 'Generating Report...' : `Download ${format === 'excel' ? 'Excel' : 'Word'} Report`}
      </button>
      
      <p className="info">
        ℹ️ Both formats include comprehensive activity logs for audit purposes
      </p>
    </div>
  );
};

export default ReportDownloader;
```
