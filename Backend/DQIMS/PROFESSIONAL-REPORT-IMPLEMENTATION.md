### Sheet 1: Summary
```java
// Create summary sheet
Sheet summarySheet = workbook.createSheet("Summary");

// Add RRA logo
addLogo(workbook, summarySheet, 0, 0);

// Add title (row 5)
Row titleRow = summarySheet.createRow(5);
Cell titleCell = titleRow.createCell(0);
titleCell.setCellValue("RWANDA REVENUE AUTHORITY");
CellStyle titleStyle = workbook.createCellStyle();
Font titleFont = workbook.createFont();
titleFont.setBold(true);
titleFont.setFontHeightInPoints((short) 16);
titleFont.setColor(IndexedColors.BLUE.getIndex());
titleStyle.setFont(titleFont);
titleStyle.setAlignment(HorizontalAlignment.CENTER);
titleCell.setCellStyle(titleStyle);

// Merge cells for title
summarySheet.addMergedRegion(new CellRangeAddress(5, 5, 0, 5));

// Add statistics
int rowNum = 8;
addStatRow(summarySheet, rowNum++, "Total Issues:", totalIssues);
addStatRow(summarySheet, rowNum++, "Open Issues:", openIssues);
addStatRow(summarySheet, rowNum++, "In Progress:", inProgressIssues);
addStatRow(summarySheet, rowNum++, "Resolved:", resolvedIssues);
addStatRow(summarySheet, rowNum++, "Closed:", closedIssues);
```

### Sheet 2: Issues List
```java
// Create issues sheet
Sheet issuesSheet = workbook.createSheet("Issues");

// Create header row
Row headerRow = issuesSheet.createRow(0);
CellStyle headerStyle = workbook.createCellStyle();
headerStyle.setFillForegroundColor(IndexedColors.BLUE.getIndex());
headerStyle.setFillPattern(FillPatternType.SOLID_FOREGROUND);
Font headerFont = workbook.createFont();
headerFont.setColor(IndexedColors.WHITE.getIndex());
headerFont.setBold(true);
headerStyle.setFont(headerFont);
headerStyle.setBorderBottom(BorderStyle.THIN);
headerStyle.setBorderTop(BorderStyle.THIN);
headerStyle.setBorderLeft(BorderStyle.THIN);
headerStyle.setBorderRight(BorderStyle.THIN);

// Add headers
String[] headers = {"ID", "Title", "Status", "Priority", "Department", "Reported By", "Created Date"};
for (int i = 0; i < headers.length; i++) {
    Cell cell = headerRow.createCell(i);
    cell.setCellValue(headers[i]);
    cell.setCellStyle(headerStyle);
}

// Add data rows with alternating colors
CellStyle evenRowStyle = workbook.createCellStyle();
evenRowStyle.setFillForegroundColor(IndexedColors.GREY_25_PERCENT.getIndex());
evenRowStyle.setFillPattern(FillPatternType.SOLID_FOREGROUND);

int rowNum = 1;
for (Issue issue : issues) {
    Row row = issuesSheet.createRow(rowNum);
    CellStyle rowStyle = (rowNum % 2 == 0) ? evenRowStyle : null;
    
    createCell(row, 0, issue.getId().toString(), rowStyle);
    createCell(row, 1, issue.getTitle(), rowStyle);
    createCell(row, 2, issue.getStatus().name(), rowStyle);
    createCell(row, 3, issue.getPriority(), rowStyle);
    createCell(row, 4, issue.getDepartment(), rowStyle);
    createCell(row, 5, issue.getReportedBy().getName(), rowStyle);
    createCell(row, 6, issue.getCreatedAt().toString(), rowStyle);
    
    rowNum++;
}

// Auto-size columns
for (int i = 0; i < headers.length; i++) {
    issuesSheet.autoSizeColumn(i);
}
```

## 🚀 API Endpoints

### Generate Report
```http
GET /api/v1/reports/generate?format=excel
GET /api/v1/reports/generate?format=word
GET /api/v1/reports/generate?format=pdf
```

### With Filters
```http
GET /api/v1/reports/generate?format=excel&department=Finance
GET /api/v1/reports/generate?format=word&status=OPEN
GET /api/v1/reports/generate?format=excel&startDate=2024-01-01&endDate=2024-04-30
```

### Response Headers
```
Content-Type: application/vnd.openxmlformats-officedocument.spreadsheetml.sheet (Excel)
Content-Type: application/vnd.openxmlformats-officedocument.wordprocessingml.document (Word)
Content-Type: application/pdf (PDF)
Content-Disposition: attachment; filename="DQIMS-Report-2024-04-28.xlsx"
```
